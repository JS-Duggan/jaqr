program
    = { topLevelDeclaration } EOF ;

topLevelDeclaration
    = globalVariableDeclaration
    | functionDeclaration
    | updateFunctionDeclaration
    | structDeclaration
    | implDeclaration
    | invariantDeclaration ;


globalVariableDeclaration
    = "define" variableDeclaration ";" ;

variableDeclaration
    = "let" binding [ "=" expression ]
    | "state" binding "=" expression
    | "imut" [ "state" ] binding "=" expression ;

binding
    = IDENT ":" type ;


type
    = "->" type
    | typeName { arrayDimension } ;

typeName
    = primitiveType
    | IDENT ;

primitiveType
    = "int"
    | "float"
    | "char"
    | "bool"
    | "byte" ;

arrayDimension
    = "[" expression "]" ;


(* exactly one main per program (checked in semantic analysis).
   "main" is an IDENTIFIER token with lexeme "main"; the parser checks the
   lexeme after "fn", so one token of lookahead still picks the branch.
   main is reserved: it cannot be used as any other identifier. *)
functionDeclaration
    = "fn" ( mainSignature | IDENT parameterClause ":" type ) block ;

mainSignature
    = "main" "(" ")" ;

updateFunctionDeclaration
    = "update" "fn" IDENT
      parameterClause
      parameterClause
      block ;

parameterClause
    = "(" [ parameterList ] ")" ;

parameterList
    = binding { "," binding } ;


structDeclaration
    = "struct" IDENT "{" { fieldDeclaration } "}" ";" ;

fieldDeclaration
    = binding ";" ;


implDeclaration
    = "impl" IDENT "{" { methodDeclaration } "}" ";" ;

methodDeclaration
    = functionDeclaration
    | updateMethodDeclaration ;

updateMethodDeclaration
    = "update" "fn" IDENT parameterClause block ;
